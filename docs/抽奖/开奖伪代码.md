// 开奖功能半伪代码
function 开奖(开奖码, 订单号) {
    // 1. 处理开奖码格式
    // 1.1 先将开奖码转为小写（统一格式）
    处理后开奖码 = 开奖码.转小写();
    
    // 1.2 检查格式：8位且字母数字混编
    if (处理后开奖码的长度 != 8) {
        return {成功: 否, 信息: "开奖码必须是8位"};
    } else if (!是否包含字母和数字(处理后开奖码)) {  // 调用辅助方法判断是否同时有字母和数字
        return {成功: 否, 信息: "开奖码需包含小写字母和数字"};
    }
    
    // 2. 检查开奖码是否存在
    // 从app\models\lottery_codes 表查询该开奖码的记录
    开奖码记录 = 数据库查询,where("lottery_code",lottery_code) 
    if (开奖码记录 不存在) {
        return {成功: 否, 信息: "开奖码不存在"};
    }
    
    // 3. 检查订单号是否存在并获取用户ID
    // 从订单表查询该订单号的记录app\models\orders
    订单记录 = 数据库查询("order_id",'订单号');
    if (订单记录 不存在) {
        return {成功: 否, 信息: "订单号不存在"};
    }
    用户ID = 订单记录中的user_id;  // 提取订单对应的用户ID
    
    // 4. 开始开奖逻辑
    // 4.1 获取该开奖码对应表app\models\lottery_prize的奖项等级,奖品表，字段有[id,prize_name名称，prize_level等级，status=1（1是有效）]
    奖项等级 = 开奖码记录中的prize_level;
    
    // 4.2 查询奖项等级对应的奖品信息
    get_prize_level = select prize_level from  lottery_codes where lottery_code = lottery_code
    get_lottery_prize = select * from  lottery_prize where prize_level= get_prize_level and status=1  #中奖的奖品id和名称

    奖品信息 = get_lottery_prize;
    
    // 4.3 判断是否中奖（有奖品信息即为中奖）
    if (奖品信息 存在) {
        是否中奖 = 是;
    } else {
        是否中奖 = 否;
    }
    
    // 5. 写入开奖记录表
    开奖记录数据 = {
        user_id: 用户ID,
        order_no: 订单号,
        lottery_code: 处理后开奖码,
        prize_level: 奖项等级,
        prize_id: get_lottery_prize.id
        draw_time: 当前时间,
        is_won: 是否中奖
    };
    lottery_draw_records => app\models\lottery_draw_records
    数据库执行("INSERT INTO lottery_draw_records VALUES (开奖记录数据)");
    
    // 6. 标记该开奖码为禁用（status=0）
    数据库执行("UPDATE lottery_codes SET status = 0 WHERE lottery_code = '处理后开奖码'");
    
    
    // 7. 返回开奖结果
    return {
        成功: 是,
        数据: {             
            order_no: 订单号,
            lottery_code: 处理后开奖码,
            is_won: 是否中奖,
            prize_level: 奖项等级,
            prize_id:  get_lottery_prize.id
        }
    };
    
}

// 辅助方法：判断字符串是否同时包含小写字母和数字
function 是否包含字母和数字(字符串) {
    有字母 = 字符串中包含[a-z]范围内的字符;
    有数字 = 字符串中包含[0-9]范围内的字符;
    return 有字母 且 有数字;
}
