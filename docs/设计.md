  <!-- 顶部查询区域 -->
  <header class="sticky top-0 z-10 bg-gray-light p-4">
    <div class="flex items-center gap-2">
      <div class="relative flex-1">
        <input 
          type="text" 
          id="queryInput"
          placeholder="输入订单号或中奖码" 
          class="w-full h-11 px-4 rounded-lg border border-gray-medium bg-white text-gray-dark text-sm focus:outline-none focus:border-primary transition-all duration-200"
          aria-label="输入订单号或中奖码"
        >
        <button 
          id="clearBtn" 
          class="absolute right-4 top-1/2 -translate-y-1/2 text-gray-text hidden"
          aria-label="清除输入内容"
        >
          <i class="fa fa-times"></i>
        </button>
      </div>
      <button 
        id="searchBtn" 
        class="h-11 px-4 rounded-lg bg-gray-medium text-white text-sm font-medium disabled:opacity-70 press-effect"
        disabled
        aria-label="查询中奖信息"
      >
        查询
      </button>
    </div>
  </header>

  <!-- 主内容区域 -->
  <main class="pb-8">
    <!-- 加载中提示 -->
    <div id="loading" class="hidden py-12 text-center">
      <div class="inline-block animate-spin rounded-full h-6 w-6 border-b-2 border-primary mb-2"></div>
      <p class="text-gray-text text-sm">查询中，请稍候...</p>
    </div>

    <!-- 无数据提示 -->
    <div id="noData" class="py-20 text-center">
      <i class="fa fa-inbox text-6xl text-gray-medium mb-4"></i>
      <p class="text-gray-text text-sm">暂未符合条件的数据</p>
    </div>

    <!-- 错误提示 -->
    <div id="error" class="hidden py-20 text-center">
      <i class="fa fa-exclamation-circle text-6xl text-danger mb-4"></i>
      <p id="errorMsg" class="text-danger text-sm mb-4">查询失败，请重试</p>
      <button id="retryBtn" class="h-10 px-4 rounded-lg bg-primary text-white text-sm font-medium press-effect">
        重新查询
      </button>
    </div>

    <!-- 结果列表 -->
    <div id="resultList" class="px-3.5 space-y-3 hidden"></div>
  </main>

  <!-- 弱网提示 -->
  <div id="weakNetwork" class="fixed top-20 left-1/2 -translate-x-1/2 bg-warning text-white text-xs py-2 px-4 rounded-full shadow-lg hidden z-50">
    网络较慢，请耐心等待
  </div>

  <script>
    // DOM元素
    const queryInput = document.getElementById('queryInput');
    const clearBtn = document.getElementById('clearBtn');
    const searchBtn = document.getElementById('searchBtn');
    const loading = document.getElementById('loading');
    const noData = document.getElementById('noData');
    const error = document.getElementById('error');
    const errorMsg = document.getElementById('errorMsg');
    const retryBtn = document.getElementById('retryBtn');
    const resultList = document.getElementById('resultList');
    const weakNetwork = document.getElementById('weakNetwork');

    // 监听输入框变化
    queryInput.addEventListener('input', handleInputChange);
    
    // 清除按钮点击事件
    clearBtn.addEventListener('click', () => {
      queryInput.value = '';
      handleInputChange();
      queryInput.focus();
    });
    
    // 查询按钮点击事件
    searchBtn.addEventListener('click', performSearch);
    
    // 重试按钮点击事件
    retryBtn.addEventListener('click', performSearch);
    
    // 处理输入框变化
    function handleInputChange() {
      const value = queryInput.value.trim();
      
      // 显示/隐藏清除按钮
      if (value) {
        clearBtn.classList.remove('hidden');
        // 启用查询按钮
        searchBtn.disabled = false;
        searchBtn.classList.remove('bg-gray-medium');
        searchBtn.classList.add('bg-primary', 'hover:bg-primaryDark');
      } else {
        clearBtn.classList.add('hidden');
        // 禁用查询按钮
        searchBtn.disabled = true;
        searchBtn.classList.remove('bg-primary', 'hover:bg-primaryDark');
        searchBtn.classList.add('bg-gray-medium');
        searchBtn.innerHTML = '查询';
      }
    }
    
    // 执行查询
    function performSearch() {
      const query = queryInput.value.trim();
      if (!query) return;
      
      // 显示加载状态
      toggleStates('loading');
      
      // 模拟网络请求
      simulateNetworkRequest(query);
    }
    
    // 模拟网络请求
    function simulateNetworkRequest(query) {
      // 随机决定是否显示弱网提示
      if (Math.random() > 0.5) {
        showWeakNetworkToast();
      }
      
      // 模拟网络延迟
      setTimeout(() => {
        // 随机生成模拟数据 (实际项目中替换为真实API请求)
        const hasError = Math.random() < 0.2; // 20%概率模拟错误
        const hasData = !hasError && Math.random() > 0.3; // 70%概率有数据
        
        if (hasError) {
          // 模拟错误
          toggleStates('error');
          errorMsg.textContent = '查询失败，请重试';
        } else if (hasData) {
          // 模拟有数据
          const mockData = generateMockData(query);
          renderResults(mockData);
          toggleStates('result');
        } else {
          // 模拟无数据
          toggleStates('noData');
        }
      }, 1500);
    }
    
    // 生成模拟数据
    function generateMockData(query) {
      const prizes = ['手机', '平板电脑', '智能手表', '无线耳机', '充电宝', '购物卡', '优惠券', '谢谢参与'];
      const count = Math.floor(Math.random() * 3) + 1; // 1-3条数据
      const data = [];
      
      for (let i = 0; i < count; i++) {
        // 生成随机日期 (最近30天内)
        const date = new Date();
        date.setDate(date.getDate() - Math.floor(Math.random() * 30));
        
        // 格式化日期时间
        const formattedDate = `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, '0')}-${String(date.getDate()).padStart(2, '0')} ${String(date.getHours()).padStart(2, '0')}:${String(date.getMinutes()).padStart(2, '0')}:${String(date.getSeconds()).padStart(2, '0')}`;
        
        data.push({
          orderNo: query.startsWith('OD') ? query : `33830${Math.floor(Math.random() * 10000000)}`,
          lotteryCode: `LT${Math.floor(Math.random() * 100000000)}`,
          prizeName: prizes[Math.floor(Math.random() * prizes.length)],
          lotteryTime: formattedDate
        });
      }
      
      return data;
    }
    
    // 渲染查询结果
    function renderResults(data) {
      resultList.innerHTML = '';
      
      data.forEach(item => {
        const card = document.createElement('div');
        card.className = 'bg-white rounded-xl border border-gray-medium p-4 card-shadow hover:card-shadow transition-all duration-200 press-effect';
        card.innerHTML = `
          <div class="grid grid-cols-2 gap-3 mb-4">
            <div>
              <span class="text-gray-text text-sm">订单号：</span>
              <span class="text-gray-dark text-sm">${item.orderNo}</span>
            </div>
            <div>
              <span class="text-gray-text text-sm">开奖码：</span>
              <span class="text-gray-dark text-sm">${item.lotteryCode}</span>
            </div>
          </div>
          <div class="grid grid-cols-2 gap-3">
            <div>
              <span class="text-gray-text text-sm">奖品名称：</span>
              <span class="text-gray-dark text-sm font-medium">${item.prizeName}</span>
            </div>
            <div>
              <span class="text-gray-text text-sm">中奖时间：</span>
              <span class="text-gray-dark text-sm">${item.lotteryTime}</span>
            </div>
          </div>
        `;
        resultList.appendChild(card);
      });
    }
    
    // 切换不同状态的显示
    function toggleStates(state) {
      loading.classList.add('hidden');
      noData.classList.add('hidden');
      error.classList.add('hidden');
      resultList.classList.add('hidden');
      
      switch(state) {
        case 'loading':
          loading.classList.remove('hidden');
          searchBtn.innerHTML = '<i class="fa fa-spinner fa-spin mr-1"></i> 加载中';
          break;
        case 'noData':
          noData.classList.remove('hidden');
          searchBtn.innerHTML = '查询';
          break;
        case 'error':
          error.classList.remove('hidden');
          searchBtn.innerHTML = '查询';
          break;
        case 'result':
          resultList.classList.remove('hidden');
          searchBtn.innerHTML = '查询';
          break;
      }
    }
    
    // 显示弱网提示
    function showWeakNetworkToast() {
      weakNetwork.classList.remove('hidden');
      setTimeout(() => {
        weakNetwork.classList.add('hidden');
      }, 3000);
    }
    
    // 初始状态设置
    document.addEventListener('DOMContentLoaded', () => {
      queryInput.focus();
    });
  </script>
</body>
